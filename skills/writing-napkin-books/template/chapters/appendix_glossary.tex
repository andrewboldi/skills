\chapter{Glossary}
\label{app:glossary}

\begin{description}[leftmargin=!,labelwidth=3.4cm,font=\sffamily\bfseries\color{navy},itemsep=1.5pt]
  \item[Half-life] The time after which half of a decaying quantity is left; $t_{1/2}=\tau\ln 2$ (\cref{ch:tour}).
  \item[Time constant] The time $\tau$ in which an exponentially decaying quantity falls to $1/e$, about
        \qty{37}{\percent}, of its value (\cref{ch:tour}).
\end{description}
